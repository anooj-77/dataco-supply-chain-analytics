## YOY calculation Measures
 
orders YOY% = 
VAR CY = [Total Orders]
VAR PY = [Last Year orders]
VAR _perc = DIVIDE(CY-PY,PY)
VAR _format = FORMAT( _perc,"0.0%;0.0%")
RETURN
IF(
    _perc>0,
    UNICHAR(9650) & " " & _format,
    UNICHAR(9660) & " " & _format)
